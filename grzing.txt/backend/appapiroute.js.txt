import {prisma} from "@/lib/prisma";
import { NextResponse } from "next/server";
import { validarCli } from "@/lib/validaciones";

export async function GET() {
    try{
        const clientes = await prisma.cliente.findMany({
            orderBy:{
                id: "asc",
            },
        });
        return NextResponse.json(clientes);
    }catch(error){
        return NextResponse.json(
            {mensaje: "Error al obtener clientes", error: error.message},
            {status: 500}
        );
    }
}

export async function POST(request) {
    try{
        const body = await request.json();
        const errores = validarCli(body);
        if(errores.length > 0){
            return NextResponse.json(
                {mensaje: "Datos invalidos", errores},
                {status: 400}
            );
        }
        const exisCedula = await prisma.cliente.findUnique({
            where: {
                cedula: body.cedula
            }
        });
        if (exisCedula){
            return NextResponse.json(
                {mensaje: "cedula ya registrada"},
                {status: 409}
            );
        }
        const cliente = await prisma.cliente.create(
            {data: body}
        );
        return NextResponse.json(cliente,{
            status: 201
        });
    }catch(error){
        return NextResponse.json(
            {mensaje: error.message},
            {status:500}
    );
    }
}
