import { Prisma } from "@/lib/Prisma";
import { NextResponse } from "next/server";
export async function GET(request, {params}) {
    try{
        const pedido = await prisma.pedido.findUnique({
            where:{
                id: Number(params.id)
            },
            include:{
                cliente: true, detalles:{include:{producto:true}}
            }
        });
        if(!pedido){
            return NextResponse.json(
                {mensaje:"pedido no encontrado"},
                {status: 404}
            );
        }
        return NextResponse.json(pedido);
    }catch(error){
        return NextResponse.json(
            {mensaje:"Error al obtener pedido", error: error.message},
            {status:500}
        );
    }
}
export async function PUT() {
    return NextResponse.json(
        {mensaje:"no permitido"},
        {status: 405}
    );
}

export async function DELETE(request, {params}) {
    try{
        const id = Number(params.id);
        const pedido = await prisma.pedido.findUnique({
            where:{id},
            include:{detalles:true}
        });
        if(!pedido){
            return NextResponse.json(
                {mensaje:"Pedido no encontrado"},
                {status:404}
            );
        }
        await prisma.$transaction(async(tx)=>{
            for(const detalles of pedido.detalles){
                const producto = await tx.producto.findUnique({
                    where:{id: detalle.productoId}
                });
                await tx.producto.update({
                    where:{id: detalle.productoId},
                    data:{stock: producto.stock + detalle.cantidad}
                });
            }
            await tx.detallePedido.deleteMany({
                where:{pedidoId: id}
            });
            await tx.pedido.delete({
                where:{id}
            });
        });
        return NextResponse.json(
            {mensaje:"pedido eliminado"}
        );
    }catch(error){
        return NextResponse.json(
            {mensaje:"Error al eliminar", error: error.message},
            {status:500}
        );
    }
}
